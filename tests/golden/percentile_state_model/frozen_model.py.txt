def adjusted_orderings(source, period, inclusive, equal_mode, remove_mode, nan_mode, tail_anchor,
                       right_equal_stop=False):
    ordered = []
    results = []
    for i, item in enumerate(source):
        number = math.nan if item is None else item
        expired = None
        if i >= period:
            old_item = source[i-period]
            indices = [j for j,(at,value) in enumerate(ordered)
                       if (at == i-period if (nan_mode if old_item is None else remove_mode) == 'id'
                           else math.isnan(value) if old_item is None else value == old_item)]
            mode = nan_mode if old_item is None else remove_mode
            if indices:
                expired = indices[-1 if mode=='last' else 0]
        old = math.nan
        if expired is not None:
            _, old = ordered.pop(expired)
        if math.isnan(number):
            pos = len(ordered)
        elif expired is None or math.isnan(old):
            pos = next((j for j,(_,value) in enumerate(ordered)
                        if (number <= value if inclusive else number < value)),len(ordered))
            if tail_anchor and pos == len(ordered):
                finite = [j for j,(_,v) in enumerate(ordered) if not math.isnan(v)]
                if finite:
                    pos = finite[-1]+1
        else:
            pos = expired
            if number < old or (number == old and equal_mode == 'left'):
                while pos and not number > ordered[pos-1][1]:
                    pos -= 1
            elif number > old or (number == old and equal_mode == 'right'):
                while pos < len(ordered) and not (number <= ordered[pos][1] if right_equal_stop
                                                else number < ordered[pos][1]):
                    pos += 1
        ordered.insert(pos,(i,number))
        results.append([v for _,v in ordered])
    return results


def compare(ordering_fn):
    totals, failures = {}, []
    for capture_name, data in CAPTURES.items():
        cache = {}
        count = mismatch = 0
        for c, title in enumerate(data['columns']):
            if not title.startswith(('Nearest ', 'Linear ')):
                continue
            parts = title.split()
            if len(parts) == 2:
                kind, percentage = parts
                name, period = 'source', 4
            elif len(parts) == 3:
                kind, name, percentage = parts
                period = 4
            else:
                kind, name, period, percentage = parts
                period = int(period)
            percentage = int(percentage)
            if (name,period) not in cache:
                source = data['columns'].index(name)
                cache[name,period] = ordering_fn([r['values'][source] for r in data['rows']],period)
            for i, ordered in enumerate(cache[name,period]):
                if i < period-1:
                    actual = None
                elif kind == 'Nearest':
                    actual = ordered[max(0,math.ceil(period*percentage/100)-1)]
                else:
                    virtual = max(0,min(period-1,period*percentage/100-.5))
                    lower = math.floor(virtual)
                    upper = min(lower+1,period-1)
                    actual = ordered[lower]+(ordered[upper]-ordered[lower])*(virtual-lower)
                actual = None if actual is None or math.isnan(actual) else actual
                expected = data['rows'][i]['values'][c]
                count += 1
                if actual != expected and not (actual is not None and expected is not None and
                                               math.isclose(actual,expected,rel_tol=0,abs_tol=1e-8)):
                    mismatch += 1
                    failures.append(dict(capture=capture_name,title=title,index=i,official=expected,
                                         candidate=actual,ordering=[None if math.isnan(v) else v for v in ordered]))
        totals[capture_name] = dict(cells=count,differences=mismatch)
    return dict(totals=totals,differences=sum(v['differences'] for v in totals.values()),failures=failures)
